2. Generators

💡 function* + yield. .next() → {value, done}.
function* countUp(){ yield 1; yield 2; yield 3; }
const g = countUp();
g.next(); // {value:1, done:false}

function* fib(){ let a=0,b=1; while(true){ yield a; [a,b]=[b,a+b]; } }
const f = fib();
f.next().value; // 0  (then 1, 1, 2...)

🎯 Generators quiz

Q1. What does `.next()` return?

{value, done}

Q2. What keyword marks a generator?

function*

5. fetch

📡 Talk to your Express API. Returns a promise; check res.ok.
async function getUsers(){
  const res = await fetch("https://api.example.com/users");
  if (!res.ok) throw new Error(`HTTP ${res.status}`);
  return await res.json();
}
async function createUser(name){
  const res = await fetch("/api/users", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ name })
  });
  if (!res.ok) throw new Error(`HTTP ${res.status}`);
  return await res.json();
}

C1 — countWords 
function countWords(text){
  const m = new Map();
  for (const w of text.split(/\s+/).filter(Boolean)) m.set(w,(m.get(w)||0)+1);
  return m; // Map
}
C2 — slugify 
const slugify = s => s.toLowerCase().replace(/[^a-z0-9]+/g,"-").replace(/^-+|-+$/g,"");
C3 — range generator 
function* range(s,e){ for(let i=s;i<=e;i++) yield i; }
C4 — fetchWithTimeout 
async function fetchWithTimeout(url,ms){
  const c=new AbortController(); const t=setTimeout(()=>c.abort(),ms);
  try{ const res=await fetch(url,{signal:c.signal}); if(!res.ok) throw new Error(`HTTP ${res.status}`); return await res.json(); }
  finally{ clearTimeout(t); }
}
C5 — freq 
function freq(text){
  const o={};
  for (const w of text.split(/\s+/).filter(Boolean)) o[w]=(o[w]||0)+1;
  return o;
}
