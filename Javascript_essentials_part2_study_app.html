1. Promises

💡 A Promise is an object representing a value that will be available now, later, or never. It starts pending, then becomes fulfilled (with a value) or rejected (with a reason).

🧪 Live Promise Simulator

Promise State: pending
Promise created — click a button.

Simple: a promise lifecycle

const fetchUser = new Promise((resolve, reject) => {
  setTimeout(() => {
    const ok = true;
    if (ok) resolve({ name: "Rohit", id: 7 });
    else reject(new Error("User not found"));
  }, 1000);
});

fetchUser
  .then((user) => console.log("Loaded:", user.name)) // "Loaded: Rohit"
  .catch((err) => console.error("Error:", err.message))
  .finally(() => console.log("Done loading")); // always runs

Chaining:

new Promise((resolve) => resolve(4))
  .then((x) => x * 2)          // 8
  .then((x) => x + 1)          // 9
  .then(console.log);          // logs 9

Promise combinators

// all — all must succeed; resolves with an array of values
const [users, posts] = await Promise.all([fetchUsers(), fetchPosts()]);

// allSettled — never rejects; reports every outcome
const results = await Promise.allSettled([fetchUsers(), fetchMissing()]);

// race — first to settle wins (great for timeouts)
function withTimeout(promise, ms) {
  const timer = new Promise((_, reject) =>
    setTimeout(() => reject(new Error("Timed out")), ms));
  return Promise.race([promise, timer]);
}

// any — first to succeed wins (fails only if all reject)
const firstOk = await Promise.any([cdnA(), cdnB(), cdnC()]);

🧠 Quick pick: all → all · every outcome → allSettled · fastest → race · first success → any

2. async / await

💡 `async function` always returns a promise. `await` pauses the function until the awaited promise settles, then unwraps its value.

🧪 Live Code Sandbox


Output will appear here…
function wait(ms) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}
async function greet() {
  await wait(500);
  return "Hello after 500ms";
}
greet().then(console.log);      // "Hello after 500ms"

try/catch with await

async function loadProfile(id) {
  try {
    const user = await fetchUser(id);   // may reject
    console.log("Profile:", user);
    return user;
  } catch (err) {
    console.error("Failed to load:", err.message);
    return null;
  }
}

Sequential vs parallel

// Sequential (slow):  timeA + timeB
const a = await taskA();
const b = await taskB();

// Parallel (fast):    max(timeA, timeB)
const [a2, b2] = await Promise.all([taskA(), taskB()]);

`for await...of` — async iteration

async function* generateNumbers() {
  yield 1; yield 2; yield 3;
}
for await (const num of generateNumbers()) {
  console.log(num); // 1, then 2, then 3
}
// Read a Node.js stream line-by-line:
for await (const line of readline.createInterface(input)) {
  console.log("Line:", line);
}

Real-world example: fetching with async/await

async function getGitHubUser(username) {
  const res = await fetch(`https://api.github.com/users/${username}`);
  if (!res.ok) throw new Error(`HTTP ${res.status}`);
  return await res.json();
}
getGitHubUser("rohithr1008")
  .then((u) => console.log("Login:", u.login))
  .catch((err) => console.error("Failed:", err.message));

Custom errors

class ValidationError extends Error {
  constructor(message) {
    super(message);
    this.name = "ValidationError";
  }
}
function validateAge(age) {
  if (age < 0) throw new ValidationError("Age cannot be negative");
}
try {
  validateAge(-5);
} catch (err) {
  if (err instanceof ValidationError) console.error("Validation:", err.message);
}

4. Classes & OOP

💡 A class is a blueprint for creating objects. It bundles a `constructor` and methods.
class User {
  constructor(name, age) {
    this.name = name;
    this.age = age;
  }
  greet() {
    return `Hi, I'm ${this.name}`;
  }
}
const rohit = new User("Rohit", 25);
console.log(rohit.greet()); // "Hi, I'm Rohit"

Getters, setters & static

class Circle {
  constructor(radius) { this._radius = radius; }
  get area() { return Math.PI * this._radius ** 2; }
  set radius(v) {
    if (v < 0) throw new RangeError("Radius can't be negative");
    this._radius = v;
  }
}
class MathUtils {
  static sum(...n) { return n.reduce((a, b) => a + b, 0); }
}
MathUtils.sum(1, 2, 3); // 6

Private fields (`#`) & inheritance

class BankAccount {
  #balance = 0;
  deposit(amount) { this.#balance += amount; }
  getBalance() { return this.#balance; }
}
class Admin extends User {
  constructor(name, age, permissions) {
    super(name, age);
    this.permissions = permissions;
  }
  can(perm) { return this.permissions.includes(perm); }
}

Worked Example: a small class hierarchy

Problem: build a `BankAccount` with private `#balance`, then a `SavingsAccount` subclass that can't go overdrawn and adds interest.

First attempt:

class BankAccount {
  // your code here — private #balance, deposit, getBalance
}
class SavingsAccount extends BankAccount {
  // your code here — withdraw guard + addInterest
}

Solution:

class BankAccount {
  #balance = 0;
  deposit(amount) { this.#balance += amount; }
  getBalance() { return this.#balance; }
}
class SavingsAccount extends BankAccount {
  constructor(rate) { super(); this.rate = rate; }
  withdraw(amount) {
    if (amount > this.getBalance()) throw new Error("Insufficient funds");
    this.deposit(-amount);
  }
  addInterest() { this.deposit(this.getBalance() * this.rate); }
}

5. Modules

📦 ES modules: `import` / `export`. Named exports use braces; default exports don't.
// math.js
export const PI = 3.14159;
export function square(n) { return n * n; }

// main.js
import { PI, square } from "./math.js";
square(4); // 16

// logger.js
export default function log(msg) { console.log(msg); }
import log from "./logger.js";

Rename & re-export & dynamic import

import { square as sq } from "./math.js";
import * as math from "./math.js";
// index.js barrel
export { PI, square } from "./math.js";
// dynamic
async function loadTheme() {
  const mod = await import("./theme.js");
  mod.apply();
}

6. Common Pitfalls

1. Forgetting to await or catch: an unhandled rejection can crash the process.
2. Believing await pauses the whole program: it only pauses its own async function.
3. Nested sequential instead of Promise.all: use `await Promise.all([a(), b()])` for parallel.
4. Losing `this`: `setTimeout(this.save, 1000)` → use `() => this.save()`.
5. Mutating a copied reference: `const b = a` shares the object — deep-copy with `structuredClone(a)`.
6. Using `==` instead of `===`: always `===` unless you need coercion.

C1 — Promise queue
async function inSequence(tasks) {
  const results = [];
  for (const task of tasks) results.push(await task());
  return results;
}
C2 — Retry with backoff
async function retry(fn, times) {
  for (let attempt = 1; attempt <= times; attempt++) {
    try { return await fn(); }
    catch (err) { if (attempt === times) throw err; await delay(attempt * 500); }
  }
}
C3 — Account hierarchy
class Account {
  constructor(owner, balance) { this.owner = owner; this.balance = balance; }
  deposit(a) { this.balance += a; }
  withdraw(a) { if (a > this.balance) throw new Error("Insufficient"); this.balance -= a; }
}
class SavingsAccount extends Account {
  constructor(owner, balance, rate) { super(owner, balance); this.rate = rate; }
  addInterest() { this.balance += this.balance * this.rate; }
}
C4 — allSettled report
async function report(promises) {
  const results = await Promise.allSettled(promises);
  return {
    succeeded: results.filter(r => r.status === "fulfilled").map(r => r.value),
    failed: results.filter(r => r.status === "rejected").map(r => r.reason)
  };
}
C5 — Dynamic import loader
async function loadMod(name) {
  try { const mod = await import(name); mod.default(); }
  catch (err) { console.error("Failed:", err.message); }
}
